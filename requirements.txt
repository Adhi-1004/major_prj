# CUDA compute libraries for ctranslate2. The NVIDIA *driver* is enough for
# ctranslate2 to report a GPU, but not to run on it -- without these, inference
# fails with "Library cublas64_12.dll is not found" and falls back to CPU.
# Harmless on machines without a GPU; the pipeline degrades to CPU either way.
# On Windows these DLLs are not on the default search path; see
# infra/cuda_runtime.py, which puts them there.
nvidia-cublas-cu12==12.9.2.10; platform_machine == "x86_64" or platform_machine == "AMD64"
nvidia-cudnn-cu12==9.24.0.43; platform_machine == "x86_64" or platform_machine == "AMD64"
